# v0.2.0 Playable Loop Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the first playable MUD loop: authenticated player creates one character, enters Blackpine Outpost, enters the Corrupt Forest, moves on a 5x5 grid, gathers resources, and sees inventory/log feedback.

**Architecture:** Implement the first thin slice of the reusable MUD engine. Shared DTOs define the command/state contract, `packages/content` defines world content, `packages/game-rules` owns map and inventory rules, `apps/server` persists authoritative state, and `apps/web` renders the MUD shell without owning gameplay rules.

**Tech Stack:** TypeScript, pnpm workspaces, Fastify, Drizzle/PostgreSQL, React/Vite, Vitest, Playwright.

---

## File Structure

- `packages/shared/src/game.ts`: public game DTOs, commands, state, class ids, location ids, item ids.
- `packages/shared/src/version.ts`: bump product and add `engineVersion`.
- `packages/content/src/world.ts`: content definitions for classes, Blackpine Outpost, Corrupt Forest, items, resources.
- `packages/game-rules/src/map-rules.ts`: pure movement, boundary, resource lookup, exit lookup.
- `packages/game-rules/src/inventory-rules.ts`: pure item stack increase/decrease.
- `packages/game-rules/src/engine.ts`: pure command reducer that combines content plus map/inventory rules.
- `apps/server/src/db/schema.ts`: add characters, inventories, map state, game events.
- `apps/server/src/modules/game/*`: server repositories, service, routes, route tests.
- `apps/web/src/features/game/*`: client API and MUD shell.
- `docs/releases/v0.2.0.md`: release note and version boundary.

## Scope Boundary

v0.2.0 implements only:

- one character per account
- Blackpine Outpost
- Corrupt Forest 5x5 map
- enter zone
- move with `W/A/S/D` or mouse buttons
- gather immediate resource
- inventory and event log display

v0.2.0 does not implement ATB combat, injury, market economy, NPC action loop, NPC AI dialogue, AI map generation, dungeons, weather, hunger, or long-running gathering.

```ts
import type { Direction, GridPositionDto, MapCellDto } from "@ai-mud/shared";
import type { ZoneDefinition } from "@ai-mud/content";

export type MoveResult =
  | { ok: true; position: GridPositionDto }
  | { ok: false; reason: "OUT_OF_BOUNDS" };

export function movePosition(
  zone: ZoneDefinition,
  current: GridPositionDto,
  direction: Direction
): MoveResult {
  const next = { ...current };
  if (direction === "north") next.y -= 1;
  if (direction === "south") next.y += 1;
  if (direction === "west") next.x -= 1;
  if (direction === "east") next.x += 1;

  if (next.x < 0 || next.y < 0 || next.x >= zone.width || next.y >= zone.height) {
    return { ok: false, reason: "OUT_OF_BOUNDS" };
  }

  return { ok: true, position: next };
}

export function buildMapCells(
  zone: ZoneDefinition,
  playerPosition: GridPositionDto,
  resourceCharges: Record<string, number>
): MapCellDto[] {
  const cells: MapCellDto[] = [];

  for (let y = 0; y < zone.height; y += 1) {
    for (let x = 0; x < zone.width; x += 1) {
      const markers: MapCellDto["markers"] = [];
      const hasPlayer = playerPosition.x === x && playerPosition.y === y;
      const hasExit = zone.exits.some((exit) => exit.position.x === x && exit.position.y === y);
      const hasResource = zone.resources.some(
        (resource) =>
          resource.position.x === x &&
          resource.position.y === y &&
          (resourceCharges[resource.id] ?? resource.charges) > 0
      );

      if (hasPlayer) markers.push("player");
      if (hasResource) markers.push("resource");
      if (hasExit) markers.push("exit");
      if (markers.length === 0) markers.push("ordinary");

      cells.push({ x, y, markers });
    }
  }

  return cells;
}
```

```ts
import type { ItemId } from "@ai-mud/shared";

export interface InventoryStack {
  itemId: ItemId;
  quantity: number;
}

export function addInventoryItem(
  inventory: InventoryStack[],
  itemId: ItemId,
  quantity: number
): InventoryStack[] {
  const existing = inventory.find((item) => item.itemId === itemId);

  if (!existing) {
    return [...inventory, { itemId, quantity }];
  }

  return inventory.map((item) =>
    item.itemId === itemId ? { ...item, quantity: item.quantity + quantity } : item
  );
}
```

```ts
  const token = request.cookies[app.config.SESSION_COOKIE_NAME];
  if (!token) {
    reply.code(401).send({ error: { code: "UNAUTHENTICATED", message: "Not signed in" } });
    return null;
  }

  const auth = new AuthService();
  const authRepo = new AuthRepository(app.di.db);
  const account = await authRepo.findAccountBySessionTokenHash(auth.hashToken(token));
  if (!account || account.status !== "active") {
    reply.code(401).send({ error: { code: "UNAUTHENTICATED", message: "Not signed in" } });
    return null;
  }

  return account;
}

export async function registerGameRoutes(app: FastifyInstance) {
  app.get("/game/state", async (request, reply) => {
    const account = await requireAccount(app, request, reply);
    if (!account) return reply;
    const service = new GameService();
    return service.getState();
  });
}
```

```ts
async function requestGame<T>(path: string, options?: RequestInit): Promise<T> {
  const response = await fetch(`${API_BASE}${path}`, {

    ...options,

    }

}
```

```tsx
import { useEffect, useMemo, useState } from "react";
import type { CharacterClassId, Direction, GameStateDto } from "@ai-mud/shared";
import {
  createCharacter,
  enterCorruptForest,
  gather,
  getGameState,
  move
} from "./gameApi";
import "./GameShell.css";

interface GameShellProps {
  csrfToken: string;
}

const initialState: GameStateDto = {
  character: null,
  locationTitle: "黑松哨站",
  locationDescription: "你尚未创建角色。",
  map: null,
  inventory: [],
  availableActions: ["create_character"],
  log: []
};

const keyDirections: Record<string, Direction> = {
  w: "north",
  a: "west",
  s: "south",
  d: "east"
};

export function GameShell({ csrfToken }: GameShellProps) {
  const [state, setState] = useState<GameStateDto>(initialState);
  const [name, setName] = useState("Zichen");
  const [classId, setClassId] = useState<CharacterClassId>("ranger");
  const [error, setError] = useState<string | null>(null);

  const canMove = state.availableActions.includes("move");
  const canGather = state.availableActions.includes("gather");
  const canEnterForest = state.availableActions.includes("enter_corrupt_forest");

  useEffect(() => {
    let cancelled = false;
    void getGameState()
      .then((nextState) => {
        if (!cancelled) setState(nextState);
      })
      .catch(() => {
        if (!cancelled) setState(initialState);
      });
    return () => {
      cancelled = true;
    };
  }, []);

  useEffect(() => {
    function onKeyDown(event: KeyboardEvent) {
      const direction = keyDirections[event.key.toLowerCase()];
      if (!direction || !canMove) return;
      event.preventDefault();
      void runCommand(() => move(direction, csrfToken));
    }

    window.addEventListener("keydown", onKeyDown);
    return () => window.removeEventListener("keydown", onKeyDown);
  }, [canMove, csrfToken]);

  async function runCommand(action: () => Promise<GameStateDto>) {
    setError(null);
    try {
      setState(await action());
    } catch {
      setError("动作失败，请稍后再试。");
    }
  }

  const cells = useMemo(() => state.map?.cells ?? [], [state.map]);

  if (!state.character) {
    return (
      <main className="game-shell">
        <section className="game-panel">
          <h1>创建角色</h1>
          <label>
            角色名
            <input value={name} onChange={(event) => setName(event.target.value)} />
          </label>
          <label>
            职业
            <select
              value={classId}
              onChange={(event) => setClassId(event.target.value as CharacterClassId)}
            >
              <option value="warrior">战士</option>
              <option value="ranger">游侠</option>
              <option value="warlock">术士</option>
            </select>
          </label>
          <button onClick={() => runCommand(() => createCharacter({ name, classId }, csrfToken))}>
            进入世界
          </button>
          {error ? <p role="alert">{error}</p> : null}
        </section>
      </main>
    );
  }

  return (
    <main className="game-shell">
      <aside className="game-panel">
        <h2>{state.character.name}</h2>
        <p>等级 {state.character.level}</p>
        <p>
          生命 {state.character.hp}/{state.character.maxHp}
        </p>
        <h3>背包</h3>
        {state.inventory.length === 0 ? <p>空</p> : null}
        {state.inventory.map((item) => (
          <p key={item.itemId}>
            {item.name} x {item.quantity}
          </p>
        ))}
      </aside>

      <section className="game-panel game-main">
        <h1>{state.locationTitle}</h1>
        <p>{state.locationDescription}</p>
        <div className="game-actions">
          {canEnterForest ? (
            <button onClick={() => runCommand(() => enterCorruptForest(csrfToken))}>
              前往腐林
            </button>
          ) : null}
          {canGather ? <button onClick={() => runCommand(() => gather(csrfToken))}>采集</button> : null}
        </div>
        {error ? <p role="alert">{error}</p> : null}
        <ol className="game-log">
          {state.log.map((entry) => (
            <li key={entry.id}>{entry.message}</li>
          ))}
        </ol>
      </section>

      <aside className="game-panel">
        <h2>地图</h2>
        <div className="mini-map">
          {cells.map((cell) => (
            <span key={`${cell.x}:${cell.y}`} className="mini-map-cell">
              {cell.markers.includes("player")
                ? "@"
                : cell.markers.includes("resource")
                  ? "R"
                  : cell.markers.includes("exit")
                    ? "E"
                    : "."}
            </span>
          ))}
        </div>
        <div className="direction-pad">
          <button disabled={!canMove} onClick={() => runCommand(() => move("north", csrfToken))}>
            W
          </button>
          <button disabled={!canMove} onClick={() => runCommand(() => move("west", csrfToken))}>
            A
          </button>
          <button disabled={!canMove} onClick={() => runCommand(() => move("south", csrfToken))}>
            S
          </button>
          <button disabled={!canMove} onClick={() => runCommand(() => move("east", csrfToken))}>
            D
          </button>
        </div>
      </aside>
    </main>
  );
}
```
